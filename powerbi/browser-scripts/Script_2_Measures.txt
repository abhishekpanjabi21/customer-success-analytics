	ref table accounts

		measure 'Latest Month' = CALCULATE ( MAX ( monthly_revenue[month] ), REMOVEFILTERS ( monthly_revenue ) )
			formatString: "mmm yyyy"
			displayFolder: "0 Helpers"

		measure 'Total MRR' =
				VAR m = MAX ( monthly_revenue[month] )
				RETURN
					CALCULATE ( SUM ( monthly_revenue[mrr] ), monthly_revenue[month] = m )
			formatString: "$#,0"
			displayFolder: "1 Revenue"

		measure 'Total ARR' = [Total MRR] * 12
			formatString: "$#,0"
			displayFolder: "1 Revenue"

		measure 'Active Customers' =
				VAR m = MAX ( monthly_revenue[month] )
				RETURN
					CALCULATE ( COUNTROWS ( monthly_revenue ), monthly_revenue[month] = m, monthly_revenue[mrr] > 0 )
			formatString: "#,0"
			displayFolder: "1 Revenue"

		measure 'Avg ARR per Customer' = DIVIDE ( [Total ARR], [Active Customers] )
			formatString: "$#,0"
			displayFolder: "1 Revenue"

		measure 'MRR 12M Ago' =
				VAR m = MAX ( monthly_revenue[month] )
				RETURN
					CALCULATE ( SUM ( monthly_revenue[mrr] ), monthly_revenue[month] = EDATE ( m, -12 ) )
			formatString: "$#,0"
			displayFolder: "1 Revenue"

		measure 'MRR Growth YoY' = DIVIDE ( [Total MRR] - [MRR 12M Ago], [MRR 12M Ago] )
			formatString: "0.0%"
			displayFolder: "1 Revenue"

		measure 'New MRR' =
				VAR firstM = CALCULATE ( MIN ( monthly_revenue[month] ), REMOVEFILTERS ( monthly_revenue ) )
				RETURN
					SUMX (
						FILTER ( monthly_revenue, monthly_revenue[month] > firstM && monthly_revenue[mrr] > 0 ),
						VAR acc = monthly_revenue[account_id]
						VAR m = monthly_revenue[month]
						VAR cur = monthly_revenue[mrr]
						VAR prev = CALCULATE ( SUM ( monthly_revenue[mrr] ), REMOVEFILTERS ( monthly_revenue ), monthly_revenue[account_id] = acc, monthly_revenue[month] = EDATE ( m, -1 ) )
						RETURN IF ( ISBLANK ( prev ), cur )
					)
			formatString: "$#,0"
			displayFolder: "1 Revenue"

		measure 'Expansion MRR' =
				VAR firstM = CALCULATE ( MIN ( monthly_revenue[month] ), REMOVEFILTERS ( monthly_revenue ) )
				RETURN
					SUMX (
						FILTER ( monthly_revenue, monthly_revenue[month] > firstM && monthly_revenue[mrr] > 0 ),
						VAR acc = monthly_revenue[account_id]
						VAR m = monthly_revenue[month]
						VAR cur = monthly_revenue[mrr]
						VAR prev = CALCULATE ( SUM ( monthly_revenue[mrr] ), REMOVEFILTERS ( monthly_revenue ), monthly_revenue[account_id] = acc, monthly_revenue[month] = EDATE ( m, -1 ) )
						RETURN IF ( prev > 0 && cur > prev, cur - prev )
					)
			formatString: "$#,0"
			displayFolder: "1 Revenue"

		measure 'Contraction MRR' =
				VAR firstM = CALCULATE ( MIN ( monthly_revenue[month] ), REMOVEFILTERS ( monthly_revenue ) )
				RETURN
					SUMX (
						FILTER ( monthly_revenue, monthly_revenue[month] > firstM && monthly_revenue[mrr] > 0 ),
						VAR acc = monthly_revenue[account_id]
						VAR m = monthly_revenue[month]
						VAR cur = monthly_revenue[mrr]
						VAR prev = CALCULATE ( SUM ( monthly_revenue[mrr] ), REMOVEFILTERS ( monthly_revenue ), monthly_revenue[account_id] = acc, monthly_revenue[month] = EDATE ( m, -1 ) )
						RETURN IF ( prev > 0 && cur < prev, prev - cur )
					)
			formatString: "$#,0"
			displayFolder: "1 Revenue"

		measure 'Churned MRR' =
				VAR firstM = CALCULATE ( MIN ( monthly_revenue[month] ), REMOVEFILTERS ( monthly_revenue ) )
				RETURN
					SUMX (
						FILTER ( monthly_revenue, monthly_revenue[month] > firstM && monthly_revenue[mrr] = 0 ),
						VAR acc = monthly_revenue[account_id]
						VAR m = monthly_revenue[month]
						VAR cur = monthly_revenue[mrr]
						VAR prev = CALCULATE ( SUM ( monthly_revenue[mrr] ), REMOVEFILTERS ( monthly_revenue ), monthly_revenue[account_id] = acc, monthly_revenue[month] = EDATE ( m, -1 ) )
						RETURN prev
					)
			formatString: "$#,0"
			displayFolder: "1 Revenue"

		measure 'Net New MRR' = [New MRR] + [Expansion MRR] - [Contraction MRR] - [Churned MRR]
			formatString: "$#,0"
			displayFolder: "1 Revenue"

		measure 'New MRR (12M)' =
				VAR lm = [Latest Month]
				RETURN
					CALCULATE ( [New MRR], monthly_revenue[month] > EDATE ( lm, -12 ), monthly_revenue[month] <= lm )
			formatString: "$#,0"
			displayFolder: "1 Revenue"

		measure 'Expansion MRR (12M)' =
				VAR lm = [Latest Month]
				RETURN
					CALCULATE ( [Expansion MRR], monthly_revenue[month] > EDATE ( lm, -12 ), monthly_revenue[month] <= lm )
			formatString: "$#,0"
			displayFolder: "1 Revenue"

		measure 'Churned MRR (12M)' =
				VAR lm = [Latest Month]
				RETURN
					CALCULATE ( [Churned MRR], monthly_revenue[month] > EDATE ( lm, -12 ), monthly_revenue[month] <= lm )
			formatString: "$#,0"
			displayFolder: "1 Revenue"

		measure 'Churned Customers' = CALCULATE ( COUNTROWS ( monthly_revenue ), monthly_revenue[mrr] = 0 )
			formatString: "#,0"
			displayFolder: "2 Retention"

		measure 'Churned Customers (12M)' =
				VAR lm = [Latest Month]
				RETURN
					CALCULATE ( [Churned Customers], monthly_revenue[month] > EDATE ( lm, -12 ), monthly_revenue[month] <= lm )
			formatString: "#,0"
			displayFolder: "2 Retention"

		measure 'Customers at Start' =
				VAR m = MAX ( monthly_revenue[month] )
				RETURN
					CALCULATE ( COUNTROWS ( monthly_revenue ), monthly_revenue[month] = EDATE ( m, -1 ), monthly_revenue[mrr] > 0 )
			formatString: "#,0"
			displayFolder: "2 Retention"

		measure 'Logo Churn Rate' = DIVIDE ( [Churned Customers], [Customers at Start] )
			formatString: "0.00%"
			displayFolder: "2 Retention"

		measure 'Avg Monthly Logo Churn (12M)' =
				VAR lm = [Latest Month]
				RETURN
					AVERAGEX ( FILTER ( VALUES ( monthly_revenue[month] ), monthly_revenue[month] > EDATE ( lm, -12 ) && monthly_revenue[month] <= lm ), [Logo Churn Rate] )
			formatString: "0.00%"
			displayFolder: "2 Retention"

		measure NRR =
				VAR EndM = MAX ( monthly_revenue[month] )
				VAR StartM = EDATE ( EndM, -12 )
				VAR Cohort = CALCULATETABLE ( VALUES ( monthly_revenue[account_id] ), monthly_revenue[month] = StartM, monthly_revenue[mrr] > 0 )
				VAR StartMRR = CALCULATE ( SUM ( monthly_revenue[mrr] ), monthly_revenue[month] = StartM, Cohort )
				VAR EndMRR = CALCULATE ( SUM ( monthly_revenue[mrr] ), monthly_revenue[month] = EndM, Cohort )
				RETURN
					DIVIDE ( EndMRR, StartMRR )
			formatString: "0.0%"
			displayFolder: "2 Retention"

		measure GRR =
				VAR EndM = MAX ( monthly_revenue[month] )
				VAR StartM = EDATE ( EndM, -12 )
				VAR Cohort = CALCULATETABLE ( VALUES ( monthly_revenue[account_id] ), monthly_revenue[month] = StartM, monthly_revenue[mrr] > 0 )
				VAR StartMRR = CALCULATE ( SUM ( monthly_revenue[mrr] ), monthly_revenue[month] = StartM, Cohort )
				VAR Kept =
					SUMX (
						Cohort,
						VAR s = CALCULATE ( SUM ( monthly_revenue[mrr] ), monthly_revenue[month] = StartM )
						VAR e = COALESCE ( CALCULATE ( SUM ( monthly_revenue[mrr] ), monthly_revenue[month] = EndM ), 0 )
						RETURN IF ( e < s, e, s )
					)
				RETURN
					DIVIDE ( Kept, StartMRR )
			formatString: "0.0%"
			displayFolder: "2 Retention"

		measure 'Logo Retention' =
				VAR EndM = MAX ( monthly_revenue[month] )
				VAR StartM = EDATE ( EndM, -12 )
				VAR Cohort = CALCULATETABLE ( VALUES ( monthly_revenue[account_id] ), monthly_revenue[month] = StartM, monthly_revenue[mrr] > 0 )
				VAR Kept = COUNTROWS ( FILTER ( Cohort, CALCULATE ( SUM ( monthly_revenue[mrr] ), monthly_revenue[month] = EndM ) > 0 ) )
				RETURN
					DIVIDE ( Kept, COUNTROWS ( Cohort ) )
			formatString: "0.0%"
			displayFolder: "2 Retention"

		measure Customers = COUNTROWS ( accounts )
			formatString: "#,0"
			displayFolder: "2 Retention"

		measure 'Churn Rate' = DIVIDE ( CALCULATE ( COUNTROWS ( accounts ), accounts[status] = "Churned" ), COUNTROWS ( accounts ) )
			formatString: "0.0%"
			displayFolder: "2 Retention"

		measure NPS =
				VAR lm = [Latest Month]
				VAR r = CALCULATETABLE ( nps_responses, nps_responses[survey_date] >= EDATE ( lm, -12 ) )
				VAR total = COUNTROWS ( r )
				VAR promoters = COUNTROWS ( FILTER ( r, nps_responses[nps_score] >= 9 ) )
				VAR detractors = COUNTROWS ( FILTER ( r, nps_responses[nps_score] <= 6 ) )
				RETURN
					ROUND ( DIVIDE ( promoters - detractors, total ) * 100, 0 )
			formatString: "0"
			displayFolder: "3 Engagement"

		measure 'Seat Utilisation' =
				VAR m = MAX ( product_usage[month] )
				RETURN
					CALCULATE ( DIVIDE ( SUM ( product_usage[active_users] ), SUM ( product_usage[licensed_seats] ) ), product_usage[month] = m )
			formatString: "0.0%"
			displayFolder: "3 Engagement"

		measure 'Avg Days to Go-Live' = AVERAGE ( onboarding[days_to_go_live] )
			formatString: "0.0"
			displayFolder: "3 Engagement"

		measure 'On-time Onboarding %' =
				DIVIDE (
					CALCULATE ( COUNTROWS ( onboarding ), onboarding[onboarding_status] = "Completed On Time" ),
					CALCULATE ( COUNTROWS ( onboarding ), onboarding[onboarding_status] <> "In Progress" )
				)
			formatString: "0.0%"
			displayFolder: "3 Engagement"

		measure Tickets = COUNTROWS ( support_tickets )
			formatString: "#,0"
			displayFolder: "3 Engagement"

		measure 'Avg Resolution Hours' = AVERAGE ( support_tickets[resolution_hours] )
			formatString: "0.0"
			displayFolder: "3 Engagement"

		measure 'Avg Health Score' = AVERAGE ( account_health[health_score] )
			formatString: "0.0"
			displayFolder: "4 Health"

		measure 'Health Customers' = COUNTROWS ( account_health )
			formatString: "#,0"
			displayFolder: "4 Health"

		measure 'Health ARR' = SUM ( account_health[mrr_now] ) * 12
			formatString: "$#,0"
			displayFolder: "4 Health"

		measure 'High Risk Customers' = CALCULATE ( COUNTROWS ( account_health ), account_health[risk_tier] = "High Risk" )
			formatString: "#,0"
			displayFolder: "4 Health"

		measure 'High Risk ARR' = CALCULATE ( [Health ARR], account_health[risk_tier] = "High Risk" )
			formatString: "$#,0"
			displayFolder: "4 Health"

		measure 'Renewals at Risk' = CALCULATE ( COUNTROWS ( account_health ), account_health[risk_tier] <> "Healthy", account_health[renewal_date] < DATE ( 2027, 3, 1 ) )
			formatString: "#,0"
			displayFolder: "4 Health"

		measure 'Renewals at Risk ARR' = CALCULATE ( [Health ARR], account_health[risk_tier] <> "Healthy", account_health[renewal_date] < DATE ( 2027, 3, 1 ) )
			formatString: "$#,0"
			displayFolder: "4 Health"

		measure 'Expansion Opportunities' = COUNTROWS ( expansion_opportunities )
			formatString: "#,0"
			displayFolder: "5 Growth"

		measure 'Expansion Pipeline ARR' = SUM ( expansion_opportunities[potential_added_mrr] ) * 12
			formatString: "$#,0"
			displayFolder: "5 Growth"
